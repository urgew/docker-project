5. Create GitHub Actions workflow
---------------------------------------
Create:

.github/workflows/docker-ci.yml

name: Build and Push Docker Image

on:
  push:
    branches:
      - main
    tags:
      - "v*"

permissions:
  packages: write
  contents: read

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}/github-docker-webserver

jobs:
  docker-build:
    runs-on: ubuntu-latest

    steps:
      # 1. Checkout source code
      - name: Checkout code
        uses: actions/checkout@v4

      # 2. Log in to GitHub Container Registry
      - name: Log in to GHCR
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      # 3. Build and push development image when code is pushed to main
      - name: Build and Push Development Image
        if: startsWith(github.ref, 'refs/heads/')
        uses: docker/build-push-action@v6
        with:
          context: .
          push: true
          tags: |
            ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:${{ github.sha }}

      # 4. Build and push release image when Git tag is created
      - name: Build and Push Release Image
        if: startsWith(github.ref, 'refs/tags/v')
        uses: docker/build-push-action@v6
        with:
          context: .
          push: true
          tags: |
            ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:${{ github.ref_name }}
            ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}:latest

Step 1: Generate a GitHub Personal Access Token (PAT)
---------------------------------------------------------
       Because you are pulling from a terminal outside of GitHub Actions, your regular GitHub password will not work. You need a token.
      1. On GitHub, click your profile picture (top right) > Settings.
      2. Scroll down the left sidebar and click Developer settings > Personal access tokens > Tokens (classic).
      3. Click Generate new token (classic).
      4. Give it a note (e.g., "Docker Pull") and check the box for read:packages.
      5. Click Generate token and copy the long string of letters and numbers (it starts with ghp_).



Step 2: Log in from your Terminal
------------------------------------
      Open your Ubuntu terminal and log into the registry using your username and the token you just copied:

     echo "YOUR_GITHUB_TOKEN_HERE" | docker login ghcr.io -u urgew --password-stdin

3.  Create Kubernetes Deployment manifest
----------------------------------------------------
 Create a manifest file named deployment-webserver.yaml:

apiVersion: apps/v1
kind: Deployment
metadata:
  name: webserver-test-deployment
  namespace: webserver-test
  labels:
    app: webserver-test
spec:
  replicas: 2
  selector:
    matchLabels:
      app: webserver-test
  template:
    metadata:
      labels:
        app: webserver-test
  spec:
    containers:
    - name: web-server
      # Point this directly to your GitHub Registry URL
      image: ghcr.io/urgew/docker-project/github-docker-test1:latest
      ports:
      - containerPort: 80
      imagePullPolicy: Always

 Apply the deployment
---------------------
kubectl apply -f deployment-webserver.yaml  -n webserver-test
kubectl get pods -n webserver-test

name: Deploy to Kubernetes

on:
  workflow_dispatch:
  push:
    branches:
      - main
    paths:
      - 'kubernetes/**'
      - '.github/workflows/deploy-kubernetes.yml'

permissions:
  contents: read

jobs:
  deploy:
    runs-on: self-hosted

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Install kubectl
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          $installDir = Join-Path $env:RUNNER_TEMP "kubectl"
          New-Item -ItemType Directory -Force -Path $installDir | Out-Null

          $version = (Invoke-RestMethod `
            "https://dl.k8s.io/release/stable.txt").Trim()

          $url = "https://dl.k8s.io/release/$version/bin/windows/amd64/kubectl.exe"
          $kubectlPath = Join-Path $installDir "kubectl.exe"

          Invoke-WebRequest -Uri $url -OutFile $kubectlPath

          Add-Content -Path $env:GITHUB_PATH -Value $installDir

          & $kubectlPath version --client

      - name: Verify Kubernetes connection
        shell: powershell
        run: kubectl get nodes

      - name: Validate manifests
        shell: powershell
        run: kubectl apply --dry-run=server -f ./kubernetes/

      - name: Deploy to Kubernetes
        shell: powershell
        run: kubectl apply -f ./kubernetes/

      - name: Verify deployment
        shell: powershell
        run: |
          kubectl rollout status deployment/webserver-test-deployment `
            -n webserver-test --timeout=180s
          kubectl get deployment,service,ingress -n webserver-test
